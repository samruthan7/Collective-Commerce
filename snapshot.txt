create type public.group_status as enum
  ('open','full','bidding','locked','completed','cancelled','expired_no_offers');
create type public.commitment_level as enum ('interested','committed');

create table public.groups (
  id uuid primary key default gen_random_uuid(),
  variant_id uuid not null references public.product_variants (id) on delete restrict,
  region_id uuid not null references public.regions (id) on delete restrict,
  created_by uuid not null references public.profiles (id) on delete restrict,
  status public.group_status not null default 'open',
  min_members integer not null default 10 check (min_members > 0),
  max_members integer,
  member_count integer not null default 0 check (member_count >= 0),
  locked_member_count integer,
  join_deadline_at timestamptz not null,
  bidding_deadline_at timestamptz not null,
  selection_deadline_at timestamptz not null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  check (max_members is null or max_members >= min_members),
  check (bidding_deadline_at > join_deadline_at),
  check (selection_deadline_at > bidding_deadline_at)
);

create view public.group_members_public as

  || case when nullif(split_part(p.full_name, ' ', 2), '') is not null

-- Core: join a specific group. FOR UPDATE serialises all joins to this group.
create function public.join_group_as(
  p_group_id uuid, p_user_id uuid, p_commitment public.commitment_level
) returns public.group_members
language plpgsql security definer set search_path = '' as $$
declare
  g public.groups;
  m public.group_members;
begin
  select * into g from public.groups where id = p_group_id for update;
  if not found then raise exception 'GROUP_NOT_FOUND'; end if;
  if g.status <> 'open' or g.join_deadline_at <= now() then raise exception 'GROUP_NOT_OPEN'; end if;
  if g.max_members is not null and g.member_count >= g.max_members then raise exception 'GROUP_FULL'; end if;

  select * into m from public.group_members where group_id = p_group_id and user_id = p_user_id;
  if found and m.left_at is null then
    raise exception 'ALREADY_MEMBER';
  elsif found then
    update public.group_members
      set left_at = null, joined_at = now(), commitment = p_commitment
      where id = m.id returning * into m;
  else
    insert into public.group_members (group_id, user_id, commitment)
      values (p_group_id, p_user_id, p_commitment) returning * into m;
  end if;

  update public.groups
    set member_count = member_count + 1,
        status = case when max_members is not null and member_count + 1 >= max_members
                      then 'full'::public.group_status else status end
    where id = p_group_id;
  return m;
end $$;

-- Join the open group for a variant; create it if none; overflow happens
-- automatically because a full group is no longer "open".
create function public.join_or_create_group_as(
  p_variant_id uuid, p_region_id uuid, p_user_id uuid, p_commitment public.commitment_level
) returns public.group_members
language plpgsql security definer set search_path = '' as $$
declare
  g_id uuid;
  attempts integer := 0;
begin
  if not exists (select 1 from public.product_variants where id = p_variant_id and is_active)
     or not exists (select 1 from public.regions where id = p_region_id and is_active) then
    raise exception 'VARIANT_NOT_FOUND';
  end if;

  loop
    attempts := attempts + 1;
    g_id := null;
    select id into g_id from public.groups
      where variant_id = p_variant_id and region_id = p_region_id and status = 'open'
      for update;

    if g_id is null then
      begin
        insert into public.groups
          (variant_id, region_id, created_by, min_members, max_members,
           join_deadline_at, bidding_deadline_at, selection_deadline_at)
        values
          (p_variant_id, p_region_id, p_user_id, 10, 50,
           now() + interval '7 days', now() + interval '10 days', now() + interval '13 days')
        returning id into g_id;
      exception when unique_violation then
        if attempts >= 5 then raise exception 'GROUP_BUSY'; end if;
        continue;
      end;
    end if;

    return public.join_group_as(g_id, p_user_id, p_commitment);
  end loop;
end $$;

create function public.leave_group_as(p_group_id uuid, p_user_id uuid)
returns void
language plpgsql security definer set search_path = '' as $$
declare
  g public.groups;
  m public.group_members;
begin
  select * into g from public.groups where id = p_group_id for update;
  if not found then raise exception 'GROUP_NOT_FOUND'; end if;
  if g.status not in ('open','full','bidding') then raise exception 'LEAVE_NOT_ALLOWED'; end if;

  select * into m from public.group_members
    where group_id = p_group_id and user_id = p_user_id and left_at is null;
  if not found then raise exception 'NOT_MEMBER'; end if;

  update public.group_members set left_at = now() where id = m.id;
  update public.groups set member_count = member_count - 1 where id = p_group_id;
end $$;

-- Wrappers for logged-in users: the user id comes from the JWT, never from the client.

-- Sweeper: every transition is a guarded UPDATE, so running it twice changes nothing.
create function public.sweep_groups() returns jsonb
language plpgsql security definer set search_path = '' as $$
declare a integer; b integer; c integer; d integer;
begin
  update public.groups set status = 'bidding'
    where status in ('open','full') and join_deadline_at <= now() and member_count >= min_members;
  get diagnostics a = row_count;

  update public.groups set status = 'cancelled'
    where status = 'open' and join_deadline_at <= now() and member_count < min_members;
  get diagnostics b = row_count;

  -- Phase 6 will lock groups that have quotations; until then no offers means expired.
  update public.groups set status = 'expired_no_offers'
    where status = 'bidding' and bidding_deadline_at <= now();
  get diagnostics c = row_count;

  update public.groups set status = 'completed'
    where status = 'locked' and selection_deadline_at <= now();
  get diagnostics d = row_count;

  return jsonb_build_object('to_bidding', a, 'cancelled', b, 'expired_no_offers', c, 'completed', d);
end $$;

-- Submit or replace a quotation. Tiers: min_members ascending, price strictly falling.
create function public.submit_quotation(
  p_group_id uuid, p_delivery_days integer, p_warranty_months integer, p_notes text, p_tiers jsonb
) returns uuid language plpgsql security definer set search_path = '' as $$
declare
  g public.groups;
  q_id uuid;
  t jsonb;
  mn integer; pr bigint;
  prev_min integer := 0; prev_price bigint := null; first_min integer := null;
begin
  if (select auth.uid()) is null then raise exception 'UNAUTHENTICATED'; end if;
  if not public.is_verified_seller() then raise exception 'NOT_VERIFIED_SELLER'; end if;

  select * into g from public.groups where id = p_group_id for share;
  if not found then raise exception 'GROUP_NOT_FOUND'; end if;
  if g.status <> 'bidding' or g.bidding_deadline_at <= now() then raise exception 'BIDDING_CLOSED'; end if;

  if jsonb_typeof(p_tiers) <> 'array' or jsonb_array_length(p_tiers) not between 1 and 5 then
    raise exception 'INVALID_TIERS';
  end if;
  for t in select e.value from jsonb_array_elements(p_tiers) with ordinality as e(value, ord) order by e.ord loop
    mn := (t->>'min_members')::integer;
    pr := (t->>'unit_price_paise')::bigint;
    if mn is null or pr is null or mn < 1 or pr < 1 or mn <= prev_min
       or (prev_price is not null and pr >= prev_price) then
      raise exception 'INVALID_TIERS';
    end if;
    if first_min is null then first_min := mn; end if;
    prev_min := mn; prev_price := pr;
  end loop;
  -- The first tier must be reachable by the group's minimum.
  if first_min > g.min_members then raise exception 'INVALID_TIERS'; end if;

  insert into public.quotations (group_id, seller_id, delivery_days, warranty_months, notes)
    values (p_group_id, (select auth.uid()), p_delivery_days, p_warranty_months, nullif(trim(p_notes), ''))
  on conflict (group_id, seller_id) do update
    set status = 'submitted', delivery_days = excluded.delivery_days,
        warranty_months = excluded.warranty_months, notes = excluded.notes
  returning id into q_id;

  delete from public.quotation_tiers where quotation_id = q_id;
  insert into public.quotation_tiers (quotation_id, min_members, unit_price_paise)
    select q_id, x.min_members, x.unit_price_paise
    from jsonb_to_recordset(p_tiers) as x(min_members integer, unit_price_paise bigint);
  return q_id;
end $$;

create function public.select_offer(p_group_id uuid, p_quotation_id uuid)
returns void language plpgsql security definer set search_path = '' as $$
declare g public.groups;
begin
  if (select auth.uid()) is null then raise exception 'UNAUTHENTICATED'; end if;
  select * into g from public.groups where id = p_group_id;
  if not found then raise exception 'GROUP_NOT_FOUND'; end if;
  if g.status <> 'locked' or g.selection_deadline_at <= now() then raise exception 'SELECTION_CLOSED'; end if;

create or replace function public.sweep_groups() returns jsonb

  return jsonb_build_object('to_bidding', a, 'cancelled', b, 'locked', c,

create function public.my_orders()
returns table (
  group_id uuid, product_name text, variant_label text, seller_name text,
  seller_phone text, seller_city text, unit_price_paise bigint, delivery_days integer
)
language sql stable security definer set search_path = '' as $$
  select g.id, p.name, v.label, sp.business_name, sp.phone, sp.city,
    (select t.unit_price_paise from public.quotation_tiers t

      order by t.min_members desc limit 1),

  join public.groups g on g.id = bs.group_id
  join public.product_variants v on v.id = g.variant_id
  join public.products p on p.id = v.product_id
  join public.seller_profiles sp on sp.user_id = q.seller_id

import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";
import { getSupabaseEnv, type SupabaseEnv } from "@/lib/supabaseEnv";

/** Refreshes the login session cookie on each request. */
export async function updateSession(request: NextRequest): Promise<NextResponse> {
  let env: SupabaseEnv;
  try {
    env = getSupabaseEnv();
  } catch (error) {
    // Missing keys must not take the whole site down in development.
    console.warn(error instanceof Error ? error.message : "Supabase is not configured");
    return NextResponse.next({ request });
  }

  let response = NextResponse.next({ request });

  const supabase = createServerClient(
    env.NEXT_PUBLIC_SUPABASE_URL,
    env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll();
        },
        setAll(cookiesToSet) {
          cookiesToSet.forEach(({ name, value }) => request.cookies.set(name, value));
          response = NextResponse.next({ request });
          cookiesToSet.forEach(({ name, value, options }) =>
            response.cookies.set(name, value, options),
          );
        },
      },
    },
  );

  // Do not put any code between createServerClient and getUser.
  await supabase.auth.getUser();

  return response;
}

import { apiErrorBodySchema, type z } from "@cc/shared";
import { env } from "./env";

export class ApiError extends Error {
  readonly code: string;
  readonly status: number;

  constructor(code: string, message: string, status: number) {
    super(message);
    this.name = "ApiError";
    this.code = code;
    this.status = status;
  }
}

/**
 * GET a path from the API and validate the response with a Zod schema.
 * `path` starts with "/" (for example "/health" or "/api/v1/me").
 */
export async function apiGet<T>(
  path: string,
  schema: z.ZodType<T>,
  signal?: AbortSignal,
): Promise<T> {
  let response: Response;
  try {
    response = await fetch(`${env.NEXT_PUBLIC_API_URL}${path}`, {
      headers: { Accept: "application/json" },
      signal,
    });
  } catch {
    throw new ApiError(
      "NETWORK_ERROR",
      "Cannot reach the server. Check your connection and try again.",
      0,
    );
  }

  const payload: unknown = await response.json().catch(() => null);

  if (!response.ok) {
    const parsed = apiErrorBodySchema.safeParse(payload);
    if (parsed.success) {
      throw new ApiError(parsed.data.error.code, parsed.data.error.message, response.status);
    }
    throw new ApiError(
      "UNKNOWN_ERROR",
      `Unexpected response (${response.status}).`,
      response.status,
    );
  }

  return schema.parse(payload);
}

"use server";

import { redirect } from "next/navigation";
import { safeRedirectPath, type AuthFormState } from "@/lib/authForm";
import { loginSchema, signupSchema } from "@/lib/authSchemas";
import { createSupabaseServerClient } from "@/lib/supabase/server";

export async function loginAction(
  _previous: AuthFormState,
  formData: FormData,
): Promise<AuthFormState> {
  const parsed = loginSchema.safeParse({
    email: formData.get("email"),
    password: formData.get("password"),
  });
  if (!parsed.success) {
    return { error: parsed.error.issues[0]?.message ?? "Check your details.", notice: null };
  }

  const supabase = await createSupabaseServerClient();
  const { error } = await supabase.auth.signInWithPassword(parsed.data);
  if (error) {
    // One message for every failure, so attackers cannot tell which part was wrong.
    return { error: "Incorrect email or password.", notice: null };
  }

  redirect(safeRedirectPath(formData.get("next")));
}

export async function signupAction(
  _previous: AuthFormState,
  formData: FormData,
): Promise<AuthFormState> {
  const parsed = signupSchema.safeParse({
    fullName: formData.get("fullName"),
    email: formData.get("email"),
    password: formData.get("password"),
  });
  if (!parsed.success) {
    return { error: parsed.error.issues[0]?.message ?? "Check your details.", notice: null };
  }

  const supabase = await createSupabaseServerClient();
  const { data, error } = await supabase.auth.signUp({
    email: parsed.data.email,
    password: parsed.data.password,
    options: { data: { full_name: parsed.data.fullName } },
  });
  if (error) {
    // Server-side only: the visitor still sees the generic message.
    console.error("signup failed:", {
      code: error.code,
      status: error.status,
      message: error.message,
    });
    return {
      error: "We could not create your account. Try another email or a stronger password.",
      notice: null,
    };
  }

  // With email confirmation switched on there is no session yet.
  if (!data.session) {
    return { error: null, notice: "Check your email to confirm your account, then log in." };
  }

  redirect(safeRedirectPath(formData.get("next")));
}

export async function signOutAction(): Promise<void> {
  const supabase = await createSupabaseServerClient();
  await supabase.auth.signOut();
  redirect("/");
}

  const { data, error } = await supabase
    .from("group_members")
    .select(
      "group:groups(id, status, member_count, min_members, join_deadline_at, variant:product_variants(label, product:products(name, brand)))",
    )
    .eq("user_id", user.id)
    .is("left_at", null)
    .order("joined_at", { ascending: false });
  if (error) throw new Error("Could not load your groups.");
  const memberships = membershipSchema.parse(data);

  const { data: selections } = await supabase
    .from("buyer_selections")
    .select("group_id, quotation_id")
    .eq("user_id", user.id);

    (selections ?? []).map((s) => [s.group_id as string, s.quotation_id as string]),
  );

  const quotationIds = [...selectionByGroup.values()];
  const quoteById = new Map<
    string,
    {
      sellerId: string;
      deliveryDays: number;
      tiers: { minMembers: number; unitPricePaise: number }[];
    }
  >();
  const sellerNames = new Map<string, string>();

  if (quotationIds.length > 0) {
    const { data: quotes } = await supabase
      .from("quotations")
      .select("id, seller_id, delivery_days, quotation_tiers(min_members, unit_price_paise)")
      .in("id", quotationIds);
    for (const q of quotes ?? []) {
      quoteById.set(q.id as string, {
        sellerId: q.seller_id as string,
        deliveryDays: q.delivery_days as number,
        tiers: (q.quotation_tiers as { min_members: number; unit_price_paise: number }[]).map(
          (t) => ({
            minMembers: t.min_members,
            unitPricePaise: Number(t.unit_price_paise),
          }),
        ),
      });
    }
    const sellerIds = [...new Set([...quoteById.values()].map((q) => q.sellerId))];
    if (sellerIds.length > 0) {
      const { data: sellers } = await supabase
        .from("sellers_public")
        .select("user_id, business_name")
        .in("user_id", sellerIds);
      for (const s of sellers ?? [])
        sellerNames.set(s.user_id as string, s.business_name as string);
    }
  }

  return memberships.map(({ group }) => {
    const quoteId = selectionByGroup.get(group.id);
    const quote = quoteId ? quoteById.get(quoteId) : undefined;

      groupId: group.id,
      status: group.status,

            unitPricePaise: effectiveTier(quote.tiers, group.member_count)?.unitPricePaise ?? null,
            deliveryDays: quote.deliveryDays,
          }
        : null,
    };
  });

  const eligible = tiers.filter((t) => t.minMembers <= members);

  const upcoming = tiers.filter((t) => t.minMembers > members);
  if (upcoming.length === 0) return null;
  return upcoming.reduce((soonest, t) => (t.minMembers < soonest.minMembers ? t : soonest));
}

export function lowestIndex(prices: (number | null)[]): number {
  let best = -1;

--- packages\shared\src\money.ts
import { z } from "zod";

/**
 * Money convention (ADR-005)
 * --------------------------
 * - PostgreSQL stores every amount as BIGINT paise.
 * - In JavaScript an amount is a whole `number` of paise (a safe integer).
 * - JSON carries integers; API field names end in `Paise`.
 * - Floating-point rupees are NEVER used for calculation or storage.
 */
export type Paise = number;

/** True when the value is a whole number of paise that JavaScript can represent exactly. */
export function isPaise(value: unknown): value is Paise {
  return typeof value === "number" && Number.isSafeInteger(value);
}

/** Throws a RangeError unless the value is a safe-integer number of paise. */
export function assertPaise(value: unknown, label = "amount"): asserts value is Paise {
  if (!isPaise(value)) {
    throw new RangeError(
      `${label} must be a whole number of paise (safe integer), received: ${String(value)}`,
    );
  }
}

/**
 * Converts a value coming from outside the type system into paise.
 * Main use: the `pg` driver returns BIGINT columns as strings.
 * Rejects decimals, non-numeric text and numbers beyond the safe-integer range.
 */
export function parsePaise(value: string | number | bigint, label = "amount"): Paise {
  if (typeof value === "number") {
    assertPaise(value, label);
    return value === 0 ? 0 : value;
  }

  if (typeof value === "bigint") {
    if (value > BigInt(Number.MAX_SAFE_INTEGER) || value < BigInt(Number.MIN_SAFE_INTEGER)) {
      throw new RangeError(`${label} is outside the safe integer range: ${value.toString()}`);
    }
    return Number(value);
  }

  if (!/^-?\d+$/.test(value)) {
    throw new RangeError(`${label} must be an integer string of paise, received: "${value}"`);
  }
  const parsed = Number(value);
  assertPaise(parsed, label);
  return parsed === 0 ? 0 : parsed;
}

const RUPEE_PATTERN = /^(-)?(\d{1,13})(?:\.(\d{1,2}))?$/;

/**
 * Converts a rupee amount typed by a person (a string such as "74,500.50" or "â‚¹7,45,000")
 * into paise using integer arithmetic only. Allows at most 2 decimal places.
 */
export function rupeesToPaise(input: string): Paise {
  const cleaned = input.replace(/[â‚¹,\s]/g, "");
  const match = RUPEE_PATTERN.exec(cleaned);
  const whole = match?.[2];
  if (match === null || whole === undefined) {
    throw new RangeError(`Invalid rupee amount: "${input}"`);
  }

  const negative = match[1] === "-";
  const fraction = (match[3] ?? "").padEnd(2, "0");
  const paise = Number(whole) * 100 + Number(fraction);

  if (paise === 0) return 0;
  return negative ? -paise : paise;
}

/** Inserts commas the Indian way: last 3 digits, then groups of 2 (7,45,000). */
function groupIndianDigits(digits: string): string {
  if (digits.length <= 3) return digits;
  const lastThree = digits.slice(-3);
  const rest = digits.slice(0, -3).replace(/\B(?=(\d{2})+(?!\d))/g, ",");
  return `${rest},${lastThree}`;
}

export interface FormatINROptions {
  /**
   * "auto"   (default): show paise only when they are not zero (â‚¹74,500 or â‚¹74,500.50)
   * "always": always show two decimals (â‚¹74,500.00)
   */
  fractionDigits?: "auto" | "always";
}

/**
 * Formats paise as rupees with the rupee sign and Indian digit grouping.
 * Implemented by hand (not Intl) so output is identical on every browser and server.
 */
export function formatINR(paise: Paise, options: FormatINROptions = {}): string {
  assertPaise(paise, "paise");

  const negative = paise < 0;
  const absolute = Math.abs(paise);
  const rupees = Math.floor(absolute / 100);
  const remainder = absolute % 100;

  const showFraction = options.fractionDigits === "always" || remainder !== 0;
  const fraction = showFraction ? `.${String(remainder).padStart(2, "0")}` : "";

  return `${negative ? "-" : ""}â‚¹${groupIndianDigits(String(rupees))}${fraction}`;
}
